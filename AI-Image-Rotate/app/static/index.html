Cách Tích Hợp API Xoay 360° Vào Website Khác (Port 3978):

Bất kỳ ứng dụng web hoặc client nào cũng có thể gửi request đến http://localhost:3978/api/rotate/single (để lấy góc sau lưng 180° hoặc góc bất kỳ) hoặc /api/rotate/turntable. 

// Lấy ảnh phía sau lưng (180°) từ AI
async function getBackView(imageBase64) {
  const response = await fetch('http://localhost:3978/api/rotate/single', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      image_base64: imageBase64,
      azimuth_deg: 180.0, // 180 độ: Phía sau lưng
      elevation_deg: 0.0,
      remove_background: true
    })
  });
  
  const data = await response.json();
  const backImageBase64 = data.frames[0].image_base64;
  console.log('Ảnh sau lưng đã render:', backImageBase64);
  return backImageBase64;
}
